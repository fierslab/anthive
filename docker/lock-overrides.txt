# Explicit decisions docker/gen-lock applies on top of the scripts' PEP-723
# headers when building the shared container venv.
#
#   pkg<spec>   replace every header's spec for pkg with this one
#   -pkg        leave pkg out of the shared venv
#
# Keep every line justified; the lock is only reproducible if the reasons
# are written down.

# pyscenic (ant-aucell) pins old numpy/pandas/dask and drags in arboreto;
# it cannot co-exist with scanpy >= 1.10 in one venv. AUCell scoring is run
# from a dev checkout (its own uv env) when needed; the images leave it out.
-pyscenic
-ctxcore

# numpy: ant-ucell (and the excluded aucell/starcat headers) cap numpy<2,
# while the live ant-serve runs numpy 2.4. The services win; ant-ucell only
# uses plain numpy/pandas array ops and imports fine on numpy 2 (checked in
# the image smoke test).
numpy>=2.0,<2.5

# mcp: ant-mcp is written against the 1.x FastMCP API (mcp.server.fastmcp);
# the live server runs 1.27-1.30. 2.x is a major version: keep it out until
# ant-mcp is ported and tested.
mcp>=1.10,<2

# duckdb: ant-serve's sandbox needs allowed_paths (>= 1.3); 1.5.x is the
# tested range. Headers ask for >=0.10 which would allow anything.
duckdb>=1.3,<1.6
